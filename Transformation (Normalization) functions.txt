def transform_features_train(df):
    df = df.copy()

    # Apply Dummification
    df = dummify(df)

    # Store min and max values of target column (for inverse transformation later)
    target_min = df['SalePrice'].min()
    target_max = df['SalePrice'].max()

    # Identify numerical columns
    numeric_cols = df.select_dtypes(include=['int64', 'float64']).columns

    # Separate scalers for target variable and features
    feature_scaler = MinMaxScaler()
    target_scaler = MinMaxScaler()

    # Fit and transform feature columns (excluding SalePrice)
    feature_cols = numeric_cols.drop('SalePrice')  # Exclude target for feature scaling
    df[feature_cols] = feature_scaler.fit_transform(df[feature_cols])

    # Fit and transform SalePrice separately
    df[['SalePrice']] = target_scaler.fit_transform(df[['SalePrice']])

    return df, feature_scaler, target_min, target_max  # Return transformed data, feature scaler, min and max values of target column

    y_test_reversed = y_test * (target_max - target_min) + target_min

def transform_features_with_log_train(df):

    df = df.copy()

    # Apply Dummification
    df = dummify(df)

    # Store min and max values of target column
    target_min = df['SalePrice'].min()
    target_max = df['SalePrice'].max()
    
    # Identify numerical columns
    numeric_cols = df.select_dtypes(include=['int64', 'float64']).columns
    
    # Compute skewness and identify highly skewed columns
    skewness = df[numeric_cols].apply(lambda x: skew(x.dropna()))
    skewed_cols = skewness[skewness > 0.5].index.tolist()
    
    # Apply Log Transformation (for detected skewed columns)
    df[skewed_cols] = np.log1p(df[skewed_cols])

    # Normalize the numerical features
    scaler = MinMaxScaler()
    df[numeric_cols] = scaler.fit_transform(df[numeric_cols])

    return df, scaler, skewed_cols, target_min, target_max  # Return transformed data, fitted scaler, skewed_cols, min and max values of target column

data, scaler, skewed_cols, target_min, target_max = transform_features_with_log_train(data)

def reverse_normalization_and_log_train(y_pred, y_test, target_min, target_max):

    # Reverse Min-Max Scaling manually
    y_pred_reversed = y_pred * (target_max - target_min) + target_min
    y_test_reversed = y_test * (target_max - target_min) + target_min

    # Reversing log transformation (np.expm1 reverses np.log1p)
    y_pred_reversed = np.expm1(y_pred_reversed)
    y_test_reversed = np.expm1(y_test_reversed)

    return y_pred_reversed, y_test_reversed
-----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
y_pred_reversed, y_test_reversed = reverse_normalization_and_log_train(y_pred, y_test, target_min, target_max)
